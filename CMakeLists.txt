cmake_minimum_required(VERSION 3.21)
project(CopilotReel VERSION 0.3.3 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Copilot Reel requires Windows 11 and a Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(CopilotReel WIN32 src/main.cpp src/appearance.cpp src/renderer.cpp src/settings.cpp src/localization.cpp src/iconography.cpp src/osd.cpp src/behavior.cpp src/presets.cpp src/official_icons.cpp src/local_launch.cpp packaging/app.rc)
add_custom_command(TARGET CopilotReel POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/brands"
    "$<TARGET_FILE_DIR:CopilotReel>/assets/brands"
  COMMENT "Copying official brand artwork and provenance")
target_include_directories(CopilotReel PRIVATE src packaging)
target_compile_definitions(CopilotReel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(CopilotReel PRIVATE /utf-8 /EHsc /W4)
endif()
target_link_libraries(CopilotReel PRIVATE d3d11 d2d1 dwrite dxgi dxguid dwmapi windowsapp CoreMessaging user32 gdi32 shell32 ole32 oleaut32 propsys shlwapi comdlg32 comctl32 advapi32 wtsapi32 windowscodecs uuid imm32)
enable_testing()

target_compile_definitions(core_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
