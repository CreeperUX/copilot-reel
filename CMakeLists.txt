cmake_minimum_required(VERSION 3.21)
project(CopilotReel VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Copilot Reel requires Windows 11 and a Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(CopilotReel WIN32 src/main.cpp src/appearance.cpp src/renderer.cpp src/settings.cpp src/localization.cpp src/iconography.cpp src/osd.cpp packaging/app.rc)
target_include_directories(CopilotReel PRIVATE src packaging)
target_compile_definitions(CopilotReel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(CopilotReel PRIVATE /utf-8 /EHsc /W4)
endif()
target_link_libraries(CopilotReel PRIVATE d3d11 d2d1 dwrite dxgi dxguid dwmapi windowsapp CoreMessaging user32 gdi32 shell32 ole32 oleaut32 propsys shlwapi comdlg32 comctl32 advapi32 wtsapi32 windowscodecs uuid)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
if(MSVC)
  target_compile_options(core_tests PRIVATE /utf-8 /EHsc /W4)
endif()
add_test(NAME core COMMAND core_tests)
add_executable(settings_tests tests/settings_tests.cpp src/localization.cpp)
target_compile_definitions(settings_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(settings_tests PRIVATE user32 gdi32 comdlg32 shell32 ole32 uuid)
if(MSVC)
  target_compile_options(settings_tests PRIVATE /utf-8 /EHsc /W4)
endif()
add_test(NAME settings COMMAND settings_tests)
add_executable(launch_probe WIN32 tests/launch_probe.cpp)
target_compile_definitions(launch_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(launch_probe PRIVATE /utf-8 /EHsc /W4)
endif()
